fn greet(name:str) -> int:
  say "Hello, ", name, "!"
  ret 0

fn add(a:int, b:int) -> int:
  ret a + b

fn factorial(n:int) -> int:
  if n <= 1:
    ret 1
  ret n * factorial(n - 1)

fn main() -> int:
  x:int = add(10, 20)
  say "Sum:", x
  say "Factorial of 5:", factorial(5)
  greet("TezzNative")
  ret 0
